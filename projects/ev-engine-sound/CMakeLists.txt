cmake_minimum_required(VERSION 3.16)
project(engine_voice C)
set(CMAKE_C_STANDARD 11)
add_library(engine_voice core/engine_voice.c core/commands.c core/engine_canvas.c core/powertrain_canvas.c core/stick_controls.c)
target_include_directories(engine_voice PUBLIC core)
target_compile_options(engine_voice PRIVATE -Wall -Wextra -Werror)
target_link_libraries(engine_voice PUBLIC m)
add_executable(render_voice tools/render_voice.c)
target_link_libraries(render_voice engine_voice)
add_executable(render_powertrain tools/render_powertrain.c)
target_link_libraries(render_powertrain engine_voice)
add_executable(test_voice tests/test_voice.c)
target_link_libraries(test_voice engine_voice)
target_compile_options(test_voice PRIVATE -UNDEBUG -Wall -Wextra -Werror)
enable_testing()
add_test(NAME voice_core COMMAND test_voice)
add_executable(test_transitions tests/test_transitions.c)
target_link_libraries(test_transitions engine_voice)
target_compile_options(test_transitions PRIVATE -UNDEBUG -Wall -Wextra -Werror)
add_test(NAME transitions_32k COMMAND test_transitions)
add_library(engine_voice_16k core/engine_voice.c core/commands.c)
target_include_directories(engine_voice_16k PUBLIC core)
target_compile_definitions(engine_voice_16k PUBLIC EV_RATE=16000)
target_compile_options(engine_voice_16k PRIVATE -Wall -Wextra -Werror)
target_link_libraries(engine_voice_16k PUBLIC m)
add_executable(test_transitions_16k tests/test_transitions.c)
target_link_libraries(test_transitions_16k engine_voice_16k)
target_compile_options(test_transitions_16k PRIVATE -UNDEBUG -Wall -Wextra -Werror)
add_test(NAME transitions_16k COMMAND test_transitions_16k)
add_executable(test_stick tests/test_stick.c)
target_link_libraries(test_stick engine_voice)
target_compile_options(test_stick PRIVATE -UNDEBUG -Wall -Wextra -Werror)
add_test(NAME stick_controls_canvas COMMAND test_stick)
add_executable(test_powertrain tests/test_powertrain.c)
target_link_libraries(test_powertrain engine_voice)
target_compile_options(test_powertrain PRIVATE -UNDEBUG -Wall -Wextra -Werror)
add_test(NAME powertrain_canvas COMMAND test_powertrain)
