cmake_minimum_required(VERSION 3.16)
project(a2dp_engine_tests C)
set(CMAKE_C_STANDARD 11)
set(VOICE "${CMAKE_CURRENT_LIST_DIR}/../../../projects/ev-engine-sound")
enable_testing()
add_executable(test_boot test_boot.c ../firmware/main/controls.c)
target_include_directories(test_boot PRIVATE ../firmware/main)
target_compile_options(test_boot PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME boot_debounce COMMAND test_boot)
add_executable(test_44k "${VOICE}/tests/test_transitions.c" "${VOICE}/core/engine_voice.c" "${VOICE}/core/commands.c")
target_include_directories(test_44k PRIVATE "${VOICE}/core")
target_compile_definitions(test_44k PRIVATE EV_RATE=44100)
target_compile_options(test_44k PRIVATE -Wall -Wextra -Werror -UNDEBUG)
target_link_libraries(test_44k PRIVATE m)
add_test(NAME voice_transitions_44100 COMMAND test_44k)
